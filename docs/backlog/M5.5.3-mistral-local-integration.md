## Intent

Implement local Mistral model inference for symbolic LLM refinement in BeatForge, providing a completely offline option that maintains maximum privacy while still leveraging Mistral's LLM capabilities for MIDI refinement.

## Overview

Local Mistral model integration allows users to run Mistral's open-weight models on their own hardware, providing complete data privacy (no data ever leaves the machine) and offline capability. This is the third option in BeatForge's Mistral priority order, serving as a fallback when network access is unavailable or when users prefer complete local control.

### 1. Model Loader (`models/backends/mistral_local.py`)

```python
"""Local Mistral model loader and manager."""

import os
import torch
from pathlib import Path
from typing import Optional, Literal
from dataclasses import dataclass

from transformers import AutoModelForCausalLM, AutoTokenizer, BitsAndBytesConfig
from accelerate import dispatch_model, infer_auto_device_map


Quantization = Literal["4bit", "8bit", "16bit", "32bit", "none"]


@dataclass
class LocalMistralConfig:
    model_path: Path
    quantization: Quantization = "4bit"
    device: str = "auto"
    max_vram: Optional[int] = None  # in GB
    offload_cpu: bool = False
    trust_remote_code: bool = True


class LocalMistralLoader:
    """Loads and manages local Mistral models."""
    
    SUPPORTED_QUANTIZATIONS = {
        "4bit": {"load_in_4bit": True, "bnb_4bit_compute_dtype": torch.float16},
        "8bit": {"load_in_8bit": True},
        "16bit": {"torch_dtype": torch.float16},
        "32bit": {"torch_dtype": torch.float32},
        "none": {},
    }
    
    def __init__(self, config: LocalMistralConfig):
        self.config = config
        self._model = None
        self._tokenizer = None
    
    def load(self) -> tuple[AutoModelForCausalLM, AutoTokenizer]:
        """Load the model and tokenizer."""
        if self._model is not None:
            return self._model, self._tokenizer
        
        # Get quantization config
        quant_config = self.SUPPORTED_QUANTIZATIONS.get(
            self.config.quantization, 
            self.SUPPORTED_QUANTIZATIONS["4bit"]
        )
        
        # Create bitsandbytes config for 4-bit/8-bit
        bnb_config = None
        if self.config.quantization in ["4bit", "8bit"]:
            bnb_config = BitsAndBytesConfig(
                load_in_4bit=quant_config.get("load_in_4bit", False),
                load_in_8bit=quant_config.get("load_in_8bit", False),
                bnb_4bit_compute_dtype=quant_config.get("bnb_4bit_compute_dtype"),
                bnb_4bit_quant_type="nf4",
                bnb_4bit_use_double_quant=False,
            )
        
        # Load tokenizer
        self._tokenizer = AutoTokenizer.from_pretrained(
            str(self.config.model_path),
            trust_remote_code=self.config.trust_remote_code
        )
        
        # Load model with appropriate configuration
        model_kwargs = {
            "trust_remote_code": self.config.trust_remote_code,
            "device_map": "auto",
            "quantization_config": bnb_config,
        }
        
        # Add torch dtype if specified
        if "torch_dtype" in quant_config:
            model_kwargs["torch_dtype"] = quant_config["torch_dtype"]
        
        # Handle device
        if self.config.device != "auto":
            if self.config.device == "cpu":
                model_kwargs["device_map"] = {"": "cpu"}
            elif self.config.device.startswith("cuda"):
                model_kwargs["device_map"] = {"": self.config.device}
        
        # Load model
        self._model = AutoModelForCausalLM.from_pretrained(
            str(self.config.model_path),
            **model_kwargs
        )
        
        # Offload to CPU if requested
        if self.config.offload_cpu:
            self._offload_to_cpu()
        
        return self._model, self._tokenizer
    
    def _offload_to_cpu(self):
        """Offload model layers to CPU to free up GPU memory."""
        if self._model is None:
            return
        
        # Offload all layers except the last few
        self._model.gradient_checkpointing_enable()
        
        # Use accelerate's offloading
        device_map = infer_auto_device_map(
            self._model,
            max_memory={0: f"{self.config.max_vram}GiB"} if self.config.max_vram else None,
            offload_folder="offload",
            offload_state_dict=True
        )
        dispatch_model(self._model, device_map=device_map)
    
    def unload(self):
        """Unload model from memory."""
        if self._model is not None:
            del self._model
            self._model = None
        if self._tokenizer is not None:
            del self._tokenizer
            self._tokenizer = None
        torch.cuda.empty_cache()
    
    @property
    def is_loaded(self) -> bool:
        """Check if model is loaded."""
        return self._model is not None
```

### 2. Local Inference Client (`refine/backends/mistral_local.py`)

```python
"""Local Mistral model client for symbolic LLM refinement."""

import os
import torch
from typing import Optional
from dataclasses import dataclass
from pathlib import Path

from src.beatforge.refine.interface import SymbolicRefinementModel
from src.beatforge.models.backends.mistral_local import LocalMistralLoader, LocalMistralConfig


@dataclass
class LocalMistralRefineConfig:
    model_path: Path
    quantization: str = "4bit"
    device: str = "auto"
    max_new_tokens: int = 512
    temperature: float = 0.7
    do_sample: bool = True


class LocalMistralClient(SymbolicRefinementModel):
    """Client for local Mistral model inference."""
    
    def __init__(self, config: LocalMistralRefineConfig):
        self.config = config
        self.loader = LocalMistralLoader(LocalMistralConfig(
            model_path=config.model_path,
            quantization=config.quantization,
            device=config.device
        ))
        self._model = None
        self._tokenizer = None
    
    def refine_symbolic(self, prompt: str, context: dict) -> str:
        """
        Perform symbolic refinement using local Mistral model.
        
        Args:
            prompt: Natural language refinement instruction
            context: Symbolic context (MIDI structure, sections, etc.)
            
        Returns:
            Refined symbolic representation
        """
        # Load model if not already loaded
        if self._model is None:
            self._model, self._tokenizer = self.loader.load()
        
        # Build the input
        system_prompt = self._build_system_prompt()
        user_message = self._format_user_message(prompt, context)
        
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_message}
        ]
        
        # Format for Mistral chat template
        text = self._tokenizer.apply_chat_template(
            messages,
            tokenize=False,
            add_generation_prompt=True
        )
        
        # Tokenize
        input_ids = self._tokenizer.encode(
            text,
            return_tensors="pt",
            truncation=True,
            max_length=2048
        )
        
        # Move to device
        device = self._get_device()
        input_ids = input_ids.to(device)
        
        # Generate
        with torch.no_grad():
            output_ids = self._model.generate(
                input_ids=input_ids,
                max_new_tokens=self.config.max_new_tokens,
                temperature=self.config.temperature,
                do_sample=self.config.do_sample,
                pad_token_id=self._tokenizer.eos_token_id,
                eos_token_id=self._tokenizer.eos_token_id,
            )
        
        # Decode
        output_text = self._tokenizer.decode(
            output_ids[0][len(input_ids[0]):],
            skip_special_tokens=True
        )
        
        return self._clean_output(output_text)
    
    def _build_system_prompt(self) -> str:
        """Build system prompt for symbolic MIDI refinement."""
        return """You are a drum programming assistant. You receive symbolic MIDI 
        descriptions and natural language instructions. Your task is to 
        refine the symbolic drum patterns while maintaining musical coherence.
        
        IMPORTANT RULES:
        - Work only with symbolic data (notes, timing, velocity)
        - Never reference or imagine audio data
        - Maintain GM drum note conventions
        - Preserve channel 10 (MIDI channel 11 in 1-indexed, channel 9 in 0-indexed)
        - Output only valid MIDI modifications
        - Be concise and precise
        - Output should be parsable by MIDI processing code"""
    
    def _format_user_message(self, prompt: str, context: dict) -> str:
        """Format user message with context."""
        context_str = self._serialize_context(context)
        return f"""MIDI Symbolic Context:
{context_str}

Refinement Instruction: {prompt}

Output only the refined symbolic MIDI description in JSON format if possible."""
    
    def _get_device(self) -> torch.device:
        """Get the appropriate device for inference."""
        if self.config.device == "cpu":
            return torch.device("cpu")
        elif self.config.device.startswith("cuda"):
            return torch.device(self.config.device)
        else:
            # Auto-detect
            if torch.cuda.is_available():
                return torch.device("cuda:0")
            elif hasattr(torch.backends, "mps") and torch.backends.mps.is_available():
                return torch.device("mps")
            else:
                return torch.device("cpu")
    
    def _clean_output(self, text: str) -> str:
        """Clean up model output."""
        # Remove any leading/trailing whitespace
        text = text.strip()
        
        # Remove common model artifacts
        text = text.replace("<s>", "").replace("</s>", "")
        text = text.replace("[INST]", "").replace("[/INST]", "")
        
        return text
    
    def unload(self):
        """Unload model from memory."""
        self.loader.unload()
        self._model = None
        self._tokenizer = None
    
    @property
    def is_loaded(self) -> bool:
        """Check if model is loaded."""
        return self._model is not None
```

```python
"""Hardware detection for optimal local Mistral configuration."""

import torch
import psutil
from typing import Optional
from dataclasses import dataclass


@dataclass
class HardwareInfo:
    gpu_available: bool
    gpu_count: int
    gpu_names: list[str]
    total_vram_gb: list[float]
    available_vram_gb: list[float]
    total_ram_gb: float
    available_ram_gb: float
    mps_available: bool


class HardwareDetector:
    """Detects available hardware and recommends optimal configuration."""
    
    @staticmethod
    def detect() -> HardwareInfo:
        """Detect available hardware."""
        gpu_available = torch.cuda.is_available()
        mps_available = hasattr(torch.backends, "mps") and torch.backends.mps.is_available()
        
        # GPU info
        gpu_count = torch.cuda.device_count() if gpu_available else 0
        gpu_names = []
        total_vram_gb = []
        available_vram_gb = []
        
        if gpu_available:
            for i in range(gpu_count):
                gpu_names.append(torch.cuda.get_device_name(i))
                total_vram_gb.append(torch.cuda.get_device_properties(i).total_memory / 1e9)
                available_vram_gb.append(
                    torch.cuda.memory_allocated(i) / 1e9
                    + torch.cuda.memory_reserved(i) / 1e9
                )
        
        # RAM info
        ram = psutil.virtual_memory()
        total_ram_gb = ram.total / 1e9
        available_ram_gb = ram.available / 1e9
        
        return HardwareInfo(
            gpu_available=gpu_available,
            gpu_count=gpu_count,
            gpu_names=gpu_names,
            total_vram_gb=total_vram_gb,
            available_vram_gb=available_vram_gb,
            total_ram_gb=total_ram_gb,
            available_ram_gb=available_ram_gb,
            mps_available=mps_available
        )
    
    @staticmethod
    def recommend_config(hardware: HardwareInfo, model_size_gb: float) -> dict:
        """Recommend optimal configuration based on hardware."""
        config = {
            "quantization": "4bit",
            "device": "auto",
            "offload_cpu": False,
            "max_vram": None
        }
        
        # If no GPU, use CPU with smallest model
        if not hardware.gpu_available and not hardware.mps_available:
            config["device"] = "cpu"
            config["quantization"] = "8bit"  # 4-bit not supported on CPU
            if model_size_gb > hardware.available_ram_gb * 0.8:
                # Not enough RAM, suggest smaller model
                return {"error": "insufficient_ram", "required": model_size_gb, "available": hardware.available_ram_gb}
            return config
        
        # GPU available
        if hardware.gpu_available:
            # Calculate total available VRAM
            total_available_vram = sum(hardware.available_vram_gb)
            
            # Determine optimal quantization
            if total_available_vram >= model_size_gb * 2:
                config["quantization"] = "16bit"
            elif total_available_vram >= model_size_gb * 1.2:
                config["quantization"] = "8bit"
            else:
                config["quantization"] = "4bit"
            
            # Enable offloading if VRAM is tight
            if total_available_vram < model_size_gb * 1.5:
                config["offload_cpu"] = True
                config["max_vram"] = int(total_available_vram * 0.9)  # Leave 10% buffer
        
        # MPS (Apple Silicon)
        if hardware.mps_available and not hardware.gpu_available:
            config["device"] = "mps"
            config["quantization"] = "8bit"  # 4-bit has limited MPS support
        
        return config


# Usage example
def auto_configure_local_mistral(model_path: str) -> LocalMistralRefineConfig:
    """Auto-configure local Mistral based on available hardware."""
    hardware = HardwareDetector.detect()
    
    # Estimate model size (7B ~14GB full precision)
    model_size_gb = 14.0
    
    # Get recommendation
    recommendation = HardwareDetector.recommend_config(hardware, model_size_gb)
    
    if "error" in recommendation:
        raise RuntimeError(
            f"Insufficient resources: {recommendation['error']}. "
            f"Required: {recommendation['required']}GB, Available: {recommendation['available']}GB"
        )
    
    return LocalMistralRefineConfig(
        model_path=Path(model_path),
        quantization=recommendation["quantization"],
        device=recommendation["device"],
        max_new_tokens=512,
        temperature=0.7
    )
```

```python
class ModelDownloader:
    """Downloads and manages local Mistral models."""
    
    DEFAULT_CACHE_DIR = Path.home() / ".beatforge" / "models"
    
    def __init__(self, cache_dir: Optional[Path] = None):
        self.cache_dir = cache_dir or self.DEFAULT_CACHE_DIR
        self.api = HfApi()
    
    def download(self, model_id: str, quantization: Optional[str] = None) -> Path:
        """Download a model."""
        if model_id not in SUPPORTED_MODELS:
            raise ValueError(f"Unsupported model: {model_id}. Supported: {list(SUPPORTED_MODELS.keys())}")
        
        model_info = SUPPORTED_MODELS[model_id]
        model_path = self.cache_dir / model_id
        
        # Create cache directory if it doesn't exist
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        
        # Download with progress
        print(f"Downloading {model_info.name}...")
        
        snapshot_download(
            repo_id=model_info.huggingface_id,
            local_dir=str(model_path),
            local_dir_use_symlinks=False,
            force_download=False,
            resume_download=True,
            ignore_patterns=["*.msgpack", "*.h5", "*.tflite", "*.safetensors.index.json"],
        )
        
        print(f"Model downloaded to {model_path}")
        return model_path
    
    def list_downloaded(self) -> list[str]:
        """List downloaded models."""
        if not self.cache_dir.exists():
            return []
        
        return [
            d.name for d in self.cache_dir.iterdir()
            if d.is_dir() and (d / "config.json").exists()
        ]
    
    def get_model_info(self, model_id: str) -> ModelInfo:
        """Get information about a model."""
        if model_id not in SUPPORTED_MODELS:
            raise ValueError(f"Unknown model: {model_id}")
        return SUPPORTED_MODELS[model_id]
    
    def remove(self, model_id: str) -> None:
        """Remove a downloaded model."""
        model_path = self.cache_dir / model_id
        if model_path.exists():
            shutil.rmtree(model_path)
            print(f"Removed {model_id}")
    
    def verify(self, model_path: Path) -> bool:
        """Verify a model is properly downloaded."""
        required_files = ["config.json", "model.safetensors"]
        for file in required_files:
            if not (model_path / file).exists():
                return False
        return True
```

## Performance Optimization

### 1. Quantization Comparison

| Quantization | VRAM Usage | Quality Impact | Speed | Recommended |
|--------------|------------|----------------|-------|-------------|
| 32-bit | 28GB | None | Slow | No |
| 16-bit | 14GB | None | Medium | GPU available |
| 8-bit | 7GB | Minimal | Fast | Yes |
| 4-bit | 4GB | Small | Fast | **Yes (primary)** |

### 2. Inference Speed Estimates

| Hardware | Quantization | Tokens/sec |
|----------|--------------|------------|
| RTX 4090 (24GB) | 4-bit | 50-80 |
| RTX 4090 (24GB) | 8-bit | 40-60 |
| RTX 4090 (24GB) | 16-bit | 25-40 |
| RTX 3090 (12GB) | 4-bit | 20-30 |
| RTX 2080 Ti (11GB) | 4-bit | 15-25 |
| M1 Pro (16GB RAM) | 8-bit | 5-10 |
| CPU (16 cores) | 8-bit | 1-3 |

### 3. Memory Optimization Tips

1. **Use 4-bit quantization** for most use cases
2. **Enable layer offloading** when VRAM is limited
3. **Use smaller context windows** when possible
4. **Batch small refinements** to minimize loading/unloading
5. **Keep model loaded** for multiple operations in a session

```markdown
### Troubleshooting

**"Insufficient VRAM" Error:**
- Try 4-bit quantization: `--quantization 4bit`
- Enable CPU offloading: `--offload-cpu`
- Use a smaller model

**"CUDA out of memory" Error:**
- Reduce `--max-new-tokens`
- Use lower quantization
- Enable offloading

**Slow Performance:**
- Use GPU instead of CPU
- Ensure CUDA drivers are up to date
- Use 4-bit instead of 8-bit for better GPU utilization

**Model Download Issues:**
- Check internet connection
- Verify Hugging Face login: `huggingface-cli whoami`
- Ensure sufficient disk space (~15GB for 7B model)
```
